class Solution:
    def findMaxAverage(self, nums: List[int], k: int) -> float:
        result = -float('inf') #給result一個最小的起始值
        for i in range(len(nums) - k + 1): #使用for迴圈且次數為數字的個數扣掉要平均的個數（k）並加一，如果沒有加一則會迴圈至前一行。
            sum = 0 #給總和一個起始值
            for j in range(i, i + k): #迴圈使j=i j=i+1 j=i+2
                sum += nums[j] #得四個數字加總
            avg = sum/k
            if avg > result: 
                result = avg
        return result #回傳
